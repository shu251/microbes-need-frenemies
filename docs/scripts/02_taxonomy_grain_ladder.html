Taxonomic curation

The column SUPERGROUP_18S is close to the 18S rRNA gene tag-sequencing results to help us draw conclusions regarding the comparison. This curation is helpful for later comparison to 18S rRNA gene amplicon work, but we also need to perform manual curation. Pair taxonomic classification with full_classification. Below generates long_frenemies_annot_tax.

myzozoa <- c("Perkinsea", "Chromerida", "Colpodellida", "Colponemidia", "Apicomplexa", "Squirmida") # at phylum
class_gyrista <- c("Bacillariophyceae","Bolidophyceae","Chrysophyceae", "Dictyochophyceae", "Eustigmatophyceae", "Ochromonadales","Pelagophyceae", "Xanthophyceae")

    Class %in% class_gyrista ~ "Stramenopiles-Gyrista",
    Class %in% class_bigyra ~ "Stramenopiles-Bigyra",
    Phylum == "Synurophyceae" ~ "Stramenopiles-Gyrista",
    Phylum == "Oomycota" ~ "Stramenopiles-Gyrista",
    Phylum == "Marine_stramenopiles_MASTs group" ~ "Stramenopiles-Bigyra",
    Supergroup == "Eukaryota incertae sedis" ~ Phylum,
    Supergroup == "Haptista" ~ paste("Hacrobia", Phylum, sep = "-"),
    #
    Phylum == "Choanoflagellida" ~ "Opisthokonta-Choanoflagellata",
    #
    TRUE ~ paste(Supergroup, Phylum, sep = "-"))) %>%
  ### Continued
  mutate(SUPERGROUP_0 = case_when(
    grepl("Alveolata-", SUPERGROUP_PHYLUM) ~ SUPERGROUP_PHYLUM,
    SUPERGROUP_PHYLUM == "Unannotated eukaryote" ~ "Eukaryote-Unannotated",
    TRUE ~ str_split_i(SUPERGROUP_PHYLUM, "-", 1))) %>%
  ###
  mutate(SUPERGROUP_PHYLUM_2 = case_when(
    # Dinoflagellates
    (SUPERGROUP_PHYLUM == "Alveolata-Dinophyta" & (is.na(Class) | is.na(Order) | grepl("SG_", Order))) ~ "Dinophyta-Unannotated",
    SUPERGROUP_PHYLUM == "Alveolata-Dinophyta" ~ paste("Dinophyta-", Order, sep = ""),
    # Ciliates
    (SUPERGROUP_PHYLUM == "Alveolata-Ciliophora" & (is.na(Class) | is.na(Order) | grepl("SG_", Order))) ~ "Ciliophora-Unannotated",
    SUPERGROUP_PHYLUM == "Alveolata-Ciliophora" ~ paste("Ciliophora-", Class, sep = ""),
    # Myzozoa
    (SUPERGROUP_PHYLUM == "Alveolata-Myzozoa" & (is.na(Class) | is.na(Order) | grepl("SG_", Order))) ~ "Myzozoa-Unannotated",
    SUPERGROUP_PHYLUM == "Alveolata-Myzozoa" & Phylum %in% myzozoa ~ paste("Myzozoa-", Phylum, sep = "")
    )) %>% 
  mutate(SUPERGROUP_PHYLUM_CLASS = case_when(
    SUPERGROUP_PHYLUM == "Alveolata-Ciliophora" ~ SUPERGROUP_PHYLUM_2,
    SUPERGROUP_PHYLUM == "Alveolata-Myzozoa" ~ SUPERGROUP_PHYLUM_2,
    SUPERGROUP_PHYLUM == "Alveolata-Dinophyta" ~ SUPERGROUP_PHYLUM_2,
    TRUE ~ SUPERGROUP_PHYLUM
  ))

Taxonomy resolution ladder (grain)

For questions like Is there a decoupling in taxonomy and functional trends across taxonomic levels?, we need to evaluate if taxonomic and functional trends are consistent across different taxonomic levels/resolution/grain-levels. The goal is to determine how the taxonomy curve maps against the KO function reference. When we select a taxonomic resolution that is “cherry-picked”, we run the risk of have a less appropriate output. If the story changes when we change the taxonomic level, that can be alarming and not responsible. Ramond et al. (2025) looked at how decoupling is only credible if it survives across grain, not at one cherry-picked resolution.

Set up the grain ladder

Write a function to go through each taxonomic level.

Code 
# Same transcripts at every depth: keep rows with a resolvable lineage, rm NAs
paired_lin <- paired |> 
  filter(!is.na(full_classification))

# Grain ladder runs on the curated taxonomy (02 manual curation; 2026-10-12)
# depth 2 = curated supergroup, 3 = curated phylum (Syndiniales split from
# Dinophyta), 4 = class-level taxon inside its curated phylum, 5+ = deeper real
# ranks. See R/taxonomy_curated.R.
cur_key <- paired_lin |>
  distinct(full_classification, SUPERGROUP_0, SUPERGROUP_PHYLUM)
tax_at_depth <- function(d) {
  cur_key |>
    mutate(.tax = curated_lineage(full_classification, SUPERGROUP_0, SUPERGROUP_PHYLUM, d)) |>
    select(full_classification, .tax)
}

# deepest curated depth present
max_depth <- max(stringr::str_count(tax_at_depth(99L)$.tax, ";") + 1L)

Code 
# dat - input taxonomy matrix
# feat - input functional matrix
# label - add a label for matrix used
# Label grain - for the taxonomic level used. 
grain_fit <- function(dat, feat, label_matrix, label_grain) {
  tryCatch( # stop if error
    { # Build matrix for adonis2 fxn to run.
    bm <- build_matrix(dat, feat, min_tpm = 0, transform = params$transform)
    m  <- meta[match(labels(bm$dist), meta$sample_id), ]
    # PERMANOVA function
    perm <- run_permanova(bm$dist, m,
                          permutations = params$permutations,
                          seed = params$seed) |>
      filter(.data$term %in% c("vent_bin", "field")) |>
      select(term, R2, p)
    # dispersion check following permanova
    disp <- c(vent_bin = betadisper_check(bm$dist, m$vent_bin)$dispersion_p,
              field    = betadisper_check(bm$dist, m$field)$dispersion_p)
    #
    perm |> mutate(matrix = label_matrix, 
                   grain = label_grain,
                   n_features = ncol(bm$mat), 
                   dispersion_p = disp[.data$term],
                   skip_reason = NA_character_,
                   .before = 1)}, 
    error = function(e) {
    # if fails, stop.
    message(label_matrix, " grain ", 
            label_grain, " skipped: ", 
            conditionMessage(e))
    tibble::tibble(matrix = label_matrix, 
                   grain = label_grain, term = NA_character_,
                   R2 = NA_real_, p = NA_real_, 
                   n_features = NA_integer_,
                   dispersion_p = NA_real_, 
                   skip_reason = conditionMessage(e))
    })
}

Annotation resolution by grain

Fraction of expression (scaledTPM) whose deepest retained rank is a real taxon name rather than a placeholder. Using TPM, not distance matrices, so it is independent of the rclr/distance choice.

Code 
# RM taxa with these:
placeholder <- "Indeterminate|Unannotated|incertae|_X$|^SG_|^Eukaryote$"

resolution_by_depth <- paired_lin |>
  mutate(n_real = stringr::str_count(lineage_depth(full_classification, 99L), ";") + 1L) |>
  expand_grid(grain = 2:max_depth) |>
  summarise(
    resolved_frac_tpm = sum(scaledTPM[n_real >= grain], na.rm = TRUE) / sum(scaledTPM),
    .by = grain) |>
  left_join(
    purrr::map_dfr(2:max_depth, \(d) tibble::tibble(
      grain = d,
      n_features = n_distinct(tax_at_depth(d)$.tax))),
    by = "grain")

# Taxonomy resolution ladder  (grain)

For questions like _Is there a decoupling in taxonomy and functional trends across taxonomic levels?_, we need to evaluate if taxonomic and functional trends are consistent across different taxonomic levels/resolution/grain-levels. The goal is to determine how the taxonomy curve maps against the KO
function reference.
When we select a taxonomic resolution that is "cherry-picked", we run the risk of have a less appropriate output. If the story changes when we change the taxonomic level, that can be alarming and not responsible. Ramond et al. (2025) looked at how decoupling is only credible if it survives across grain, not at one cherry-picked resolution.

```{r grain-fit-fn}
#| message: false
#| warning: false
# dat - input taxonomy matrix
# feat - input functional matrix
# label - add a label for matrix used
# Label grain - for the taxonomic level used. 
grain_fit <- function(dat, feat, label_matrix, label_grain) {
  tryCatch( # stop if error
    { # Build matrix for adonis2 fxn to run.
    bm <- build_matrix(dat, feat, min_tpm = 0, transform = params$transform)
    m  <- meta[match(labels(bm$dist), meta$sample_id), ]
    # PERMANOVA function
    perm <- run_permanova(bm$dist, m,
                          permutations = params$permutations,
                          seed = params$seed) |>
      filter(.data$term %in% c("vent_bin", "field")) |>
      select(term, R2, p)
    # dispersion check following permanova
    disp <- c(vent_bin = betadisper_check(bm$dist, m$vent_bin)$dispersion_p,
              field    = betadisper_check(bm$dist, m$field)$dispersion_p)
    #
    perm |> mutate(matrix = label_matrix, 
                   grain = label_grain,
                   n_features = ncol(bm$mat), 
                   dispersion_p = disp[.data$term],
                   skip_reason = NA_character_,
                   .before = 1)}, 
    error = function(e) {
    # if fails, stop.
    message(label_matrix, " grain ", 
            label_grain, " skipped: ", 
            conditionMessage(e))
    tibble::tibble(matrix = label_matrix, 
                   grain = label_grain, term = NA_character_,
                   R2 = NA_real_, p = NA_real_, 
                   n_features = NA_integer_,
                   dispersion_p = NA_real_, 
                   skip_reason = conditionMessage(e))
    })
}
```

```{r save-paired}
  save_paired(paired_L4, fields = PAIRED_FIELDS, depth = PAIRED_DEPTH)
  rm(paired_L4); invisible(gc())

}
```
